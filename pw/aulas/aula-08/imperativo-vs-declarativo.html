    Um único lugar descreve COMO A TELA É, em função dos dados.

    // Lista.jsx — o componente inteiro
function Lista({ itens }) {
  // nao existe "adicionar na tela" nem
  // "atualizar o contador": a tela e
  // uma FUNCAO dos itens recebidos.

  return (
    <section>
      <p className="contador">
        {itens.length} itens na lista
      </p>

      {itens.length === 0 ? (
        <p>Nenhum item na lista.</p>
      ) : (
        <ul>
          {itens.map((item) => (
            <li key={item.id}>
              <span>{item.nome}</span>
              <span>R$ {item.preco}</span>
            </li>
          ))}
        </ul>
      )}
    </section>
  );
}
